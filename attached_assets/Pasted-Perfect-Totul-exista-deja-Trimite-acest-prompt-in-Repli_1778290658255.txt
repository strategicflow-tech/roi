downloadBtn.onclick = function() {
  var email = document.getElementById('downloadEmail').value.trim();
  var dlStatus = document.getElementById('downloadStatus');
  
  if (!email || !email.includes('@')) {
    dlStatus.textContent = 'Enter a valid email.';
    dlStatus.style.color = '#FF6B6B';
    return;
  }

  dlStatus.textContent = 'Checking access...';
  dlStatus.style.color = 'rgba(255,255,255,.4)';

  fetch('https://strategic-flow-audit.replit.app/changelog-audit/check-email', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ email: email })

  .then(function(res) { return res.json(); })

      dlStatus.textContent = 'Access confirmed. Downloading...';
      dlStatus.style.color = '#69F0AE';

      // Not a client — show upgrade message

};

**4. The `downloadHTML()` function:**

```javascript
function downloadHTML() {
  var content = document.documentElement.outerHTML;
  var blob = new Blob([content], { type: 'text/html' });
  var url = URL.createObjectURL(blob);
  var a = document.createElement('a');
  a.href = url;
  a.download = 'strategic-flow-changelog-audit.html';
  a.click();
  URL.revokeObjectURL(url);
}
```
