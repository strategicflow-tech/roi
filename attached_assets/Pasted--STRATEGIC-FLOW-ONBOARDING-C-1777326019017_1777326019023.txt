<script>
(function() {
  const TOTAL = 5;
  let current = 0, startX = 0, isDragging = false, dismissed = false;
  const track = document.getElementById('sfTrack');
  const fill = document.getElementById('sfProgressFill');
  const dotsEl = document.getElementById('sfDots');
  const prevBtn = document.getElementById('sfPrev');
  const nextBtn = document.getElementById('sfNext');
  const carousel = document.getElementById('sf-carousel');
  for (let i = 0; i < TOTAL; i++) {
    const d = document.createElement('button');
    d.className = 'sf-dot' + (i === 0 ? ' active' : '');
    d.setAttribute('aria-label', 'Slide ' + (i+1));
    d.onclick = () => sfGoTo(i);
    dotsEl.appendChild(d);
  }
  function sfGoTo(idx) {
    current = Math.max(0, Math.min(TOTAL - 1, idx));
    track.style.transform = 'translateX(-' + (current * 100) + '%)';
    fill.style.width = ((current + 1) / TOTAL * 100) + '%';
    document.querySelectorAll('.sf-dot').forEach((d, i) => d.classList.toggle('active', i === current));
    prevBtn.disabled = current === 0;
    nextBtn.disabled = current === TOTAL - 1;
  }
  window.sfNav = function(dir) { sfGoTo(current + dir); };
  window.sfDismiss = function() {
    if (dismissed) return;
    dismissed = true;
    carousel.style.maxHeight = carousel.offsetHeight + 'px';
    requestAnimationFrame(() => carousel.classList.add('dismissed'));
  };
  document.addEventListener('focusin', function(e) {
    if (!dismissed && e.target.matches('input, textarea, [contenteditable]')) sfDismiss();
  });
  const trackOuter = document.querySelector('.sf-track-outer');
  trackOuter.addEventListener('touchstart', e => { startX = e.touches[0].clientX; isDragging = true; }, { passive: true });
  trackOuter.addEventListener('touchend', e => {
    if (!isDragging) return;
    const diff = startX - e.changedTouches[0].clientX;
    if (Math.abs(diff) > 40) sfNav(diff > 0 ? 1 : -1);
    isDragging = false;
  }, { passive: true });
  trackOuter.addEventListener('mousedown', e => { startX = e.clientX; isDragging = true; });
  trackOuter.addEventListener('mouseup', e => {
    if (!isDragging) return;
    const diff = startX - e.clientX;
    if (Math.abs(diff) > 40) sfNav(diff > 0 ? 1 : -1);
    isDragging = false;
  });
  sfGoTo(0);
})();
</script>
