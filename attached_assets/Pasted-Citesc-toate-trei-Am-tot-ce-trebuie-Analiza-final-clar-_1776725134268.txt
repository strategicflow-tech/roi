```js
function isProductImage(url) {
  if (!url) return false
  const u = url.toLowerCase()
  
  // Filter by URL patterns
  const skipPatterns = [
    // Size-specific avatar crops
    'width=40', 'height=40', 'width=96', 'height=96', 'width=32', 'height=32',
    'rmode=crop',
    // Known author/avatar image IDs
    '1646653490249',
    '630c6d4e', // gravatar hash
    // Generic skip words
    'gravatar', 'avatar', 'author', 'profile', 'headshot',
    'logo', 'typelogo', 'symbol',
    'favicon',
    'keyboard-shortcuts',
    // Partner logos
    'salesforce', 'hubspot', 'google', 'microsoft', 'adobe',
    // SVG icons (usually UI icons, not product screenshots)
  ]
  
  if (skipPatterns.some(p => u.includes(p))) return false
  
  // Filter SVGs unless they are clearly product screenshots
  if (u.endsWith('.svg')) return false
  
  // Filter tiny images by URL size params
  const widthMatch = url.match(/[?&]width=(\d+)/i)
  if (widthMatch && parseInt(widthMatch[1]) < 100) return false
  
  return true
}
```

```js
function deduplicateImages(images) {
  const baseMap = new Map()
  for (const img of images) {
    const base = img.url.split('?')[0]
    const existing = baseMap.get(base)
    if (!existing) {
      baseMap.set(base, img)
    } else {
      // Keep the one with larger width param, or the original without params
      const existingWidth = parseInt((existing.url.match(/width=(\d+)/i) || [])[1] || '0')
      const newWidth = parseInt((img.url.match(/width=(\d+)/i) || [])[1] || '0')
      if (newWidth > existingWidth) baseMap.set(base, img)
    }
  }
  return Array.from(baseMap.values())
}
```
